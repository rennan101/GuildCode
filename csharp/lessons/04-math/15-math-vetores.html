15. Vetores e Operacoes com Vector3

Módulo 4 — Matemática 3D

Vector3 e a estrutura mais usada no Unity — representa uma posicao, direcao ou velocidade em 3D. Dominar vetores e essencial para programacao de jogos.

15.1 — Criando Vetores

using UnityEngine;

public class VetoresBasico : MonoBehaviour
{
    void Start()
    {
        // Criando Vector3
        Vector3 posicao = new Vector3(3f, 5f, 7f);
        Vector3 frente = Vector3.forward;  // (0, 0, 1)
        Vector3 cima = Vector3.up;         // (0, 1, 0)
        Vector3 direita = Vector3.right;   // (1, 0, 0)
        Vector3 zero = Vector3.zero;       // (0, 0, 0)
        Vector3 um = Vector3.one;          // (1, 1, 1)

        Debug.Log($"Posicao: {posicao}");
        Debug.Log($"Frente: {frente}");

        // Acessar componentes individuais
        float x = posicao.x;
        float y = posicao.y;
        float z = posicao.z;
    }
}

15.2 — Operacoes Matematicas com Vetores

void Start()
{
    Vector3 a = new Vector3(1f, 2f, 3f);
    Vector3 b = new Vector3(4f, 5f, 6f);

    // Soma
    Vector3 soma = a + b; // (5, 7, 9)

    // Subtracao
    Vector3 sub = b - a; // (3, 3, 3)

    // Multiplicacao por escalar
    Vector3 mult = a * 2f; // (2, 4, 6)

    // Distancia entre dois pontos
    float dist = Vector3.Distance(a, b);
    Debug.Log($"Distancia: {dist}"); // 5.19

    // Magnitude (comprimento do vetor)
    float mag = a.magnitude;
    Debug.Log($"Magnitude: {mag}"); // 3.74

    // Normalizar (magnitude = 1, mantem direcao)
    Vector3 dir = a.normalized;
    Debug.Log($"Normalizado: {dir}");
    Debug.Log($"Magnitude apos normalizar: {dir.magnitude}"); // 1.0
}

15.3 — Dot Product e Cross Product

void Start()
{
    Vector3 fwd = Vector3.forward;
    Vector3 right = Vector3.right;

    // Dot Product: indica se vetores apontam na mesma direcao
    // 1 = mesma direcao, 0 = perpendiculares, -1 = opostas
    float dot = Vector3.Dot(fwd, right);
    Debug.Log($"Dot (90 graus): {dot}"); // 0

    Vector3 fwd2 = Vector3.forward;
    float dot2 = Vector3.Dot(fwd, fwd2);
    Debug.Log($"Dot (0 graus): {dot2}"); // 1

    // Cross Product: retorna vetor perpendicular a ambos
    Vector3 cross = Vector3.Cross(Vector3.forward, Vector3.right);
    Debug.Log($"Cross: {cross}"); // (0, -1, 0) = para baixo
}
Quando usar cada um:
• Dot Product: "esta o jogador olhando para mim?" (dot > 0 = sim)
• Cross Product: "esta a esquerda ou direita?" (y > 0 = esquerda)

// 50% do caminho
Vector3 meio = Vector3.Lerp(posicaoA, posicaoB, 0.5f);
